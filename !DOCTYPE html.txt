        function solveGaussSeidel() {
            const input = document.getElementById('matrixInput').value.trim();
            const tol = parseFloat(document.getElementById('tolerance').value);
            const maxIter = parseInt(document.getElementById('maxIter').value);
            const errorBox = document.getElementById('errorBox');
            const resultBox = document.getElementById('resultBox');
            const outputContent = document.getElementById('outputContent');

            errorBox.style.display = 'none';
            resultBox.style.display = 'none';

            if (!input) { showError("Error: Input matrix cannot be blank."); return; }

            // Parse text grid into nested values
            let rows = input.split('\n').map(r => r.trim().split(/\s+/).map(Number));
            let n = rows.length;

            // Validation: Confirm uniform matrix grids
            for(let i=0; i<n; i++) {
                if(rows[i].length !== n + 1) {
                    showError(`Error: Row ${i+1} has invalid dimensions (${rows[i].length} columns found). For ${n} equations, expected ${n+1} columns.`);
                    return;
                }
                if(rows[i].some(isNaN)) {
                    showError(`Error: Row ${i+1} contains non-numeric character elements.`);
                    return;
                }
            }

            // Extract values into matrix framework A and vector B
            let A = [];
            let B = [];
            for(let i=0; i<n; i++) {
                A.push(rows[i].slice(0, n));
                B.push(rows[i][n]);
                if(A[i][i] === 0) {
                    showError(`Error: Zero found on primary diagonal element A[${i+1}][${i+1}]. Gauss-Seidel division cannot continue.`);
                    return;
                }
            }

            // Gauss-Seidel Calculation loop
            let X = new Array(n).fill(0);
            let iterations = 0;
            let converged = false;

            while(iterations < maxIter) {
                let xOld = [...X];
                for(let i=0; i<n; i++) {
                    let sum = 0;
                    for(let j=0; j<n; j++) {
                        if(j !== i) sum += A[i][j] * X[j];
                    }
                    X[i] = (B[i] - sum) / A[i][i];
                }

                // Check convergence status
                let maxDiff = 0;
                for(let i=0; i<n; i++) {
                    maxDiff = Math.max(maxDiff, Math.abs(X[i] - xOld[i]));
                }

                iterations++;
                if(maxDiff < tol) { converged = true; break; }
                
                // Catch exploding systems early to signify infinite loop / non-convergence
                if(maxDiff > 1e10) {
                    showError(`Infinite Iteration Warning: Matrix diverges dynamically. Values are compounding endlessly. Ensure your matrix setup stays strictly diagonally dominant.`);
                    return;
                }
            }

            if(!converged) {
                showError(`Execution Cap Met: System did not converge within ${maxIter} rounds. Infinite looping prevented.`);
                return;
            }

            // Present results dynamically
            let htmlResult = `<p><strong>Converged successfully in ${iterations} execution cycles.</strong></p><ul>`;
            for(let i=0; i<n; i++) {
                htmlResult += `<li><strong>X<sub>${i+1}</sub></strong> = ${X[i].toFixed(5)}</li>`;
            }
            htmlResult += "</ul>";
            outputContent.innerHTML = htmlResult;
            resultBox.style.display = 'block';
        }

        function showError(msg) {
            const errorBox = document.getElementById('errorBox');
            errorBox.innerText = msg;
            errorBox.style.display = 'block';
        }
